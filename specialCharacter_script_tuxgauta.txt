// ==UserScript==
// @name         CRE 4.6 Independent Special Character Search
// @namespace    cre.special.char.search.multi.v46
// @version      4.6
// @description  CRE special-character search with CSV
// @author       Tushar Gautam- @tuxgauta
// @match        https://content-risk-engine-iad.iad.proxy.amazon.com/rule-management/search*
// @grant        none
// ==/UserScript==

(function () {
  'use strict';

  const PANEL_ID = 'cre-special-search-panel';

  /* ---------------- NORMALIZATION ---------------- */
  function normalizeKeyword(k) {
    return (k || '')
      .toLowerCase()
      .replace(/\\\\/g, '\\')
      .replace(/\\(.)/g, '$1')
      .replace(/[^a-z0-9]/g, '');
  }

  function parseInputKeywords(input) {
    return input.split(/[\n,]+/).map(k => k.trim()).filter(Boolean);
  }
   function canonicalExact(k) {
  return (k || '')
    .toLowerCase()
    .replace(/\\/g, '')// remove escape backslashes
    .replace(/\s+/g, '');// remove spaces only
}

  /* ---------------- KEYWORD EXTRACTION ---------------- */
  function extractKeywordsFromRule(rule) {
    const found = [];
    rule?.conditions?.forEach(c =>
      c?.arguments?.forEach(a => {
        if (typeof a.value === 'string' && a.value.startsWith('[')) {
          try { JSON.parse(a.value).forEach(k => found.push(k)); } catch {}
        }
      })
    );
    return found;
  }

  /* ---------------- UI ---------------- */
  function injectUI() {
    if (document.getElementById(PANEL_ID)) return;

    const panel = document.createElement('div');
    panel.id = PANEL_ID;
    panel.style.cssText = `
      background:#0f172a;color:white;padding:8px;border-radius:6px;
      font-size:12px;min-width:260px;max-width:280px;
      box-shadow:0 0 8px rgba(0,0,0,.4);position:relative;
    `;

    panel.innerHTML = `
      <div id="cre-drag-handle" style="font-weight:bold;cursor:move">
        Special Char Search
      </div>
      <textarea id="cre-keyword"
        placeholder="Enter keywords (comma or new line)"
        style="width:100%;height:42px;margin:4px 0"></textarea>
      <label style="font-size:11px;display:flex;align-items:center;gap:4px">
  <input type="checkbox" id="cre-exact-match"/>
  Exact match (special characters)
</label>
      <div style="display:flex;gap:4px">
        <button id="cre-search-btn" style="flex:1">Search</button>
        <button id="cre-csv-btn" style="flex:1">CSV</button>
      </div>
      <div id="cre-status" style="margin-top:4px;opacity:.75"></div>
      <div id="cre-results" style="margin-top:6px;max-height:220px;overflow:auto"></div>
    `;

    const heading = [...document.querySelectorAll('h1,h2,div')]
      .find(e => e.textContent?.trim() === 'Search Rules');

    (heading?.parentElement || document.body).appendChild(panel);
    enableDrag(panel);

    document.getElementById('cre-search-btn').onclick = runSearch;
    document.getElementById('cre-csv-btn').onclick = exportCSV;
  }

  function enableDrag(panel) {
    const h = panel.querySelector('#cre-drag-handle');
    let d=false,x=0,y=0;
    h.onmousedown=e=>{d=true;const r=panel.getBoundingClientRect();
      x=e.clientX-r.left;y=e.clientY-r.top;
      panel.style.position='fixed';panel.style.zIndex=10000;};
    document.onmousemove=e=>d&&(panel.style.left=`${e.clientX-x}px`,panel.style.top=`${e.clientY-y}px`);
    document.onmouseup=()=>d=false;
  }

  /* ---------------- XHR INTERCEPT ---------------- */
  let searchResolve=null;
  (function(){
    const o=XMLHttpRequest.prototype.open,s=XMLHttpRequest.prototype.send;
    XMLHttpRequest.prototype.open=function(m,u,...r){this._u=u;return o.call(this,m,u,...r)};
    XMLHttpRequest.prototype.send=function(...a){
      this.addEventListener('load',()=>{
        if(this._u?.includes('/rule-management/search')){
          try{
            const d=JSON.parse(this.responseText);
            searchResolve?.(d.matchingRules||[]);
            document.getElementById('cre-status').innerText=
              `CRE returned ${d.matchingRules?.length||0} rules`;
          }catch{}
        }
      });
      return s.apply(this,a);
    };
  })();

  /* ---------------- SEARCH ---------------- */
  let matchedResults=[];

  async function runSearch() {
    const input=parseInputKeywords(document.getElementById('cre-keyword').value);
    if(!input.length) return alert('Enter keyword');

    document.getElementById('cre-results').innerHTML='';
    document.getElementById('cre-status').innerText='Searching…';

    const rules=await new Promise(r=>{
      searchResolve=r;
      [...document.querySelectorAll('button')]
        .find(b=>b.innerText?.toUpperCase().includes('SEARCH'))?.click();
    });

    const targets=input.map(normalizeKeyword);

    matchedResults=rules.map(r=>{
      const kws=extractKeywordsFromRule(r.rule);
      const exactMode =
  document.getElementById('cre-exact-match')?.checked;

const m = kws.filter(k => {
  const nk = normalizeKeyword(k);
  const rawK = k.replace(/\\/g, ''); // remove escape only

  return targets.some(t => {
   if (exactMode) {
  const inputCanon = canonicalExact(
    document.getElementById('cre-keyword').value
  );
  const ruleCanon = canonicalExact(k);

  return ruleCanon === inputCanon;
}

    // fuzzy mode (existing behavior)
    return nk.includes(t) || t.includes(nk);
  });
});
      if(!m.length) return null;

      return {
        rule:r.rule,
        matchedKeywords:m,
        marketplace:r.rule.marketplace||'UNKNOWN',
        author:r.rule.author||''
      };
    }).filter(Boolean);

    renderResults();
  }

  /* ---------------- RENDER ---------------- */
  function renderResults() {
    const el=document.getElementById('cre-results');
    if(!matchedResults.length){el.innerHTML='<em>No matches</em>';return;}

    const g={};
    matchedResults.forEach(r=>(g[r.marketplace]??=[]).push(r));

    el.innerHTML=Object.entries(g).map(([mp,rs])=>`
      <div>
        <div class="mp-header" style="cursor:pointer;color:#22d3ee;font-weight:bold">
          ▶ ${mp} (${rs.length})
        </div>
        <div class="mp-body" style="display:none;margin-left:6px">
          ${rs.map(r=>`
            <div style="margin-bottom:6px;border-bottom:1px solid #334155">
              <a target="_blank" style="color:#38bdf8"
                 href="https://content-risk-engine-iad.iad.proxy.amazon.com/rule-management/view/${r.rule.id}">
                ${r.rule.name}
              </a>
              <ul style="margin:2px 0 6px 14px;font-size:11px">
                ${r.matchedKeywords.map(k=>`<li>${k}</li>`).join('')}
              </ul>
            </div>
          `).join('')}
        </div>
      </div>`).join('');

    document.querySelectorAll('.mp-header').forEach(h=>{
      h.onclick=()=>{
        const b=h.nextElementSibling;
        const open=b.style.display==='block';
        b.style.display=open?'none':'block';
        h.textContent=h.textContent.replace(open?'▼':'▶',open?'▶':'▼');
      };
    });
  }

  /* ---------------- CSV (CLEAN) ---------------- */
  function exportCSV() {
    if(!matchedResults.length) return alert('No results');

    let csv =
      'RuleID,RuleName,Marketplace,Process,RuleAuthor,MatchedKeywords,RuleLink\n';

    matchedResults.forEach(r => {
      const ruleLink =
        `https://content-risk-engine-iad.iad.proxy.amazon.com/rule-management/view/${r.rule.id}`;

      csv +=
        `${r.rule.id},"${r.rule.name}",${r.marketplace},` +
        `"${r.rule.processes.join('|')}","${r.author}",` +
        `"${r.matchedKeywords.join(' | ')}",${ruleLink}\n`;
    });

    const a=document.createElement('a');
    a.href=URL.createObjectURL(new Blob([csv],{type:'text/csv'}));
    a.download='cre_special_keyword_results.csv';
    a.click();
  }

  injectUI();
})();
